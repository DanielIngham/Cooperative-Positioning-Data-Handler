134inline void setDataSet(int argc, char *argv[], DataHandler &data) {
135
136 std::string dataset;
137 std::string output_directory = DataHandlerDefaults::kOutputDir;
138 double sample_period = DataHandlerDefaults::kSamplePeriod;
139
140 if (argc == 2) {
141 throw std::runtime_error("No dataset provided. " +
142 helpMessage(argv[EXECUTABLE]));
143 }
144
145 int dataset_number = argToInt(argv[ARG]);
146
147 /* There are only 9 UTIAS datasets. */
148 if (dataset_number < 1 || dataset_number > 9) {
149 throw std::runtime_error("Select a dataset between 1 and 9.");
150 }
151
152 dataset = "MRCLAM_Dataset" + std::string(argv[ARG]);
153
154 for (int i = 3; i < argc; i += 2) {
155
156 if (!checkFlag(argv[i])) {
157 return;
158 }
159
160 switch (argv[i][1]) {
161
162 /* Output Directory */
163 case 'o':
164 output_directory = std::string(argv[i + 1]);
165 break;
166
167 /* Sample Period */
168 case 'p':
169 sample_period = argToDouble(argv[i + 1]);
170 break;
171
172 default:
173 throw std::runtime_error("Unknown option: " + std::string(argv[i]));
174 break;
175 }
176 }
177
178 std::cout << "\033[1;32m" << "Dataset Options" << "\033[0m" << std::endl
179 << '\t' << "- Dataset: " << dataset << std::endl
180 << '\t' << "- Sample Period: " << sample_period << std::endl
181 << '\t' << "- Output Directory: " << output_directory << std::endl
182 << std::endl;
