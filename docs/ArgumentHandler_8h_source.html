139inline void setDataSet(int argc, char *argv[], Data::Handler &data) {
140
141 std::string dataset;
142 std::string output_directory = Data::HandlerDefaults::kOutputDir;
143 double sample_period = Data::HandlerDefaults::kSamplePeriod;
144
145 if (argc == 2) {
146 throw std::runtime_error("No dataset provided. " +
147 helpMessage(argv[EXECUTABLE]));
148 }
149
150 int dataset_number = argToInt(argv[ARG]);
151
152 /* There are only 9 UTIAS datasets. */
153 if (dataset_number < 1 || dataset_number > 9) {
154 throw std::runtime_error("Select a dataset between 1 and 9.");
155 }
156
157 dataset = "MRCLAM_Dataset" + std::string(argv[ARG]);
158
159 for (int i = 3; i < argc; i += 2) {
160
161 if (!checkFlag(argv[i])) {
162 return;
163 }
164
165 switch (argv[i][1]) {
166
167 /* Output Directory */
168 case 'o':
169 output_directory = std::string(argv[i + 1]);
170 break;
171
172 /* Sample Period */
173 case 'p':
174 sample_period = argToDouble(argv[i + 1]);
175 break;
176
177 default:
178 throw std::runtime_error("Unknown option: " + std::string(argv[i]));
179 break;
180 }
181 }
182
183 /* NOTE: If the output directory is not specified by the user, the set the
184 * output directory to the executables name. */
185 if (output_directory == Data::HandlerDefaults::kOutputDir) {
186 output_directory = std::string(argv[EXECUTABLE]);
187 }
188
189 std::cout << "\033[1;32m" << "Dataset Options" << "\033[0m" << std::endl
190 << '\t' << "- Dataset: " << dataset << std::endl
191 << '\t' << "- Sample Period: " << sample_period << std::endl
192 << '\t' << "- Output Directory: " << output_directory << std::endl
193 << std::endl;
194
195 data.setDataSet(dataset, output_directory, sample_period);
196}

286inline void setArguments(int argc, char *argv[], Data::Handler &data) {
287
288 /* Check if an argument is provided. */

314 case 's':
315 setSimulation(argc, argv, data);
316 break;

320 }
