235inline std::string setPlotSettings(const PlotSettings &settings) {
236 std::ostringstream oss;
237
238 if (settings.style == BOXES) {
239 oss << " using " << settings.x << ":" << settings.y << ":"
240 << settings.box_width << " ";
241 } else if (!settings.math_expression) {
242 oss << " using " << settings.x << ":" << settings.y << " ";
243 }
244
245 oss << "title \"" << settings.title << "\" "
246 << "with " << to_string(settings.style);
247
248 if (settings.style == POINTS || settings.style == LINESPOINTS) {
249 oss << " pointtype " << to_string(settings.pointtype) << " pointsize "
250 << settings.pointsize;
251 }
252
253 if (settings.linecolor != NONE) {
254 oss << " linecolor rgb \"" << to_string(settings.linecolor) << "\" ";
255 }
256
257 oss << " linewidth " << settings.linewidth;
258
259 return oss.str();
260}

269inline std::string setAxisSettings(AxisSettings settings) {
270 std::ostringstream oss;
271
272 /* Set the x and y range if the user has changed them form default. */
273 if (settings.x_range.min != settings.x_range.max) {
274
275 /* If the min is larger than the max, switch them around. */
276 if (settings.x_range.min > settings.x_range.max) {
277 double temp = settings.x_range.max;
278 settings.x_range.max = settings.x_range.min;
279 settings.x_range.min = temp;
280 }
281
282 oss << "set xrange [" << settings.x_range.min << ":" << settings.x_range.max
283 << "]" << '\n';
284 }
285
286 if (settings.y_range.min != settings.y_range.max) {
287
288 /* If the min is larger than the max, switch them around. */
289 if (settings.y_range.min > settings.y_range.max) {
290 double temp = settings.x_range.max;
291 settings.y_range.max = settings.y_range.min;
292 settings.y_range.min = temp;
293 }
294
295 oss << "set yrange [" << settings.y_range.min << ":" << settings.y_range.max
296 << "]" << '\n';
297 }
298
299 oss << "set xlabel \"" << settings.x_label << '"' << '\n';
300 oss << "set ylabel \"" << settings.y_label << '"' << '\n';
301
302 oss << '\n';
303
304 return oss.str();
305}
